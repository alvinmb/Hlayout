# 1. The command bar (bottom of the GUI) -- type a line, press Enter
RECT 0 0,0 100,50
SREF VIA_Layer1_to_Layer3 200,0

# 2. Python, headless or via the built-in Script Console (name: hl)
from hlayout.scripting.api import Session
s = Session.new("MYLIB")
s.new_cell("TOP")
s.rect(layer=0, x1=0, y1=0, x2=100, y2=50)
s.save("out.gds")
